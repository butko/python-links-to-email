# ──────── Настройки SMTP ────────
[smtp]
EMAIL = recipient@example.com
SMTP_SERVER = smtp.yandex.ru
SMTP_PORT = 587
SMTP_USER = sender@yandex.ru
SMTP_PASSWORD = your_app_password

# ──────── Общий токен VK API ────────
# Используется всеми источниками, у которых не задан свой api_token
[vk]
api_token = ваш_токен_доступа
api_version = 5.199

# ──────── Обычная HTML-страница ────────
[source_site1]
url = https://example.com
filters = /news/, /articles/

# ──────── Второй обычный сайт без фильтра ────────
[source_site2]
url = https://another-site.com
# filters не указан — будут отправлены все ссылки

# ──────── Канал ВКонтакте (использует общий токен) ────────
[source_vk1]
url = https://vk.ru/vk_community
filters = wall-

# ──────── Ещё один канал ВКонтакте ────────
[source_vk2]
url = https://vk.ru/another_community
filters = wall-, clip-, video-

# ──────── Канал ВКонтакте со своим токеном ────────
[source_vk3]
url = https://vk.ru/private_community
api_token = свой_токен_для_этого_сообщества
api_version = 5.199
filters = wall-
